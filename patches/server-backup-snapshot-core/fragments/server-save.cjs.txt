app.post('/api/backup/server/save', async (req, res, next) => {
    if (!await checkAuth(req, res)) { return; }
    if (!checkActiveSession(req, res)) return;

    const abortTracker = createBackupSourceAbortTracker(req, res);
    let backupSource = null;
    let writeStream = null;
    let writeFailure = null;
    let tmpPath = null;

    try {
        backupSource = await capturePointInTimeBackupSource({
            includeInlays: true,
            signal: abortTracker.signal,
        });
        await waitAtBackupSourceTestGate('server-save', abortTracker.signal);
        const pinned = backupSource.snapshot;

        const pinnedInlays = backupSource.filesystemEntries
            .filter((entry) => entry.backupName.startsWith('inlay/'));
        const pinnedSidecars = backupSource.filesystemEntries
            .filter((entry) => entry.backupName.startsWith('inlay_sidecar/'));
        const namespacedEntries = [
            ...pinned.kvListWithSizes('assets/').map((entry) => ({
                kind: 'kv',
                key: entry.key,
                backupName: path.basename(entry.key),
                sortKey: entry.key,
                size: entry.size,
            })),
            ...listColdStorageBackupEntries({
                reader: pinned,
                migrateLegacy: false,
            }),
            ...pinned.kvListWithSizes('inlay_meta/').map((entry) => ({
                kind: 'kv',
                key: entry.key,
                backupName: entry.key,
                sortKey: entry.key,
                size: entry.size,
            })),
            ...pinnedInlays,
            ...pinnedSidecars,
        ];
        const pinnedDbSize = pinned.kvSize('database/database.bin');
        if (pinnedDbSize === null) {
            throw new BackupSourceError(
                'BACKUP_DATABASE_MISSING',
                'database.bin is missing from the pinned backup source',
                500,
            );
        }
        const namespacedArchiveBytes = totalBackupFramedSize(namespacedEntries);
        const totalEntries = namespacedEntries.length + 1; // +1 for database
        // Like the native counter, totalBytes leaves out database.risudat.
        const totalBytes = namespacedEntries.reduce((sum, entry) => sum + entry.size, 0);

        // Same reassembly as /api/backup/export: plugin storage and
        // deactivated characters ride inside database.risudat. The value is
        // built before the disk check so the check covers the bytes that are
        // actually written; the re-encoded database can be much larger than
        // a compressed pinned blob. A build failure (for example a missing
        // archive payload) is reported after the namespaced entries, at the
        // point where the database entry is written.
        let dbValue = null;
        let dbBuildError = null;
        try {
            dbValue = await buildPinnedFullExportDbValue(pinned);
        } catch (error) {
            dbBuildError = error;
        }
        if (!dbBuildError && !dbValue) {
            throw new BackupSourceError(
                'BACKUP_DATABASE_MISSING',
                'database.bin is missing from the pinned backup source',
                500,
            );
        }
        const databaseEntry = dbValue ? {
            kind: 'buffer',
            buffer: dbValue,
            backupName: 'database.risudat',
            sortKey: 'database.risudat',
            size: dbValue.length,
        } : null;
        const archiveBytes = databaseEntry
            ? namespacedArchiveBytes + backupEntryFramedSize(databaseEntry)
            : null;

        // Pre-flight disk check against the framed archive size. When the
        // database value could not be built, the native estimate terms are
        // used instead: the entries above, the database blob, and the plugin
        // and archive rows that database.risudat re-embeds. A missing statfs
        // implementation remains non-fatal.
        try {
            let estimate;
            if (archiveBytes !== null) {
                estimate = archiveBytes;
            } else {
                estimate = namespacedArchiveBytes + pinnedDbSize;
                for (const entry of pinned.kvListWithSizes(pluginStorage.PREFIX)) {
                    if (entry.key !== pluginStorage.MIGRATED_MARKER_KEY) estimate += entry.size;
                }
                for (const entry of pinned.kvListWithSizes(ARCHIVE_PREFIX)) estimate += entry.size;
            }
            const required = Math.ceil(estimate * 1.05);
            const statfs = await fs.statfs(backupsDir);
            const free = statfs.bsize * statfs.bavail;
            if (estimate > 0 && free < required) {
                res.status(400).json({
                    error: 'Insufficient disk space (need ~'
                        + (required / 1024 / 1024).toFixed(0)
                        + ' MB, free '
                        + (free / 1024 / 1024).toFixed(0)
                        + ' MB)',
                    code: 'insufficient_space',
                    required,
                    free,
                });
                return;
            }
        } catch (error) {
            console.warn('[Backup] pre-flight disk check failed:', error?.message || error);
        }

        res.setHeader('content-type', 'application/x-ndjson');
        res.flushHeaders();

        const paths = allocatePointInTimeServerBackupPaths();
        tmpPath = paths.tmpPath;
        const { finalPath, filename } = paths;
        const { createWriteStream: createFsWriteStream } = require('fs');
        writeStream = createFsWriteStream(tmpPath, { flags: 'wx' });
        const recordWriteFailure = (error) => { writeFailure = error; };
        writeStream.on('error', recordWriteFailure);

        const writeArchiveEntry = async (entry, value) => {
            const ok = writeStream.write(encodeBackupEntry(entry.backupName, value));
            if (!ok) await waitForBackupDrain(writeStream, abortTracker.signal);
            if (writeFailure) throw writeFailure;
        };

        let bytesWritten = 0;
        for (let index = 0; index < namespacedEntries.length; index++) {
            throwIfBackupSourceAborted(abortTracker.signal);
            if (writeFailure) throw writeFailure;
            const entry = namespacedEntries[index];
            const value = await readPointInTimeBackupEntry(
                backupSource,
                entry,
                abortTracker.signal,
            );
            await writeArchiveEntry(entry, value);
            bytesWritten += value.length;
            const written = index + 1;
            if (written % 50 === 0 || written === namespacedEntries.length) {
                res.write(JSON.stringify({
                    type: 'progress',
                    current: written,
                    total: totalEntries,
                    bytes: bytesWritten,
                    totalBytes,
                }) + '\n');
            }
        }

        throwIfBackupSourceAborted(abortTracker.signal);
        if (dbBuildError) throw dbBuildError;
        if (!Number.isSafeInteger(archiveBytes)) {
            throw new BackupSourceError(
                'BACKUP_SOURCE_TOO_LARGE',
                'Backup archive exceeds the safe byte range',
                413,
            );
        }
        await writeArchiveEntry(databaseEntry, dbValue);
        bytesWritten += dbValue.length;
        res.write(JSON.stringify({
            type: 'progress',
            current: totalEntries,
            total: totalEntries,
            bytes: bytesWritten,
            totalBytes,
        }) + '\n');

        await finishBackupWritable(writeStream, abortTracker.signal);
        writeStream.removeListener('error', recordWriteFailure);
        if (writeFailure) throw writeFailure;
        if (writeStream.bytesWritten !== archiveBytes) {
            throw new BackupSourceError(
                'BACKUP_ARCHIVE_SIZE_MISMATCH',
                'Completed server backup size does not match its pinned plan',
                500,
            );
        }
        throwIfBackupSourceAborted(abortTracker.signal);

        // Atomic rename: the file is published once the rename completes and
        // stays published even if the client disconnects afterwards.
        await fs.rename(tmpPath, finalPath);
        tmpPath = null;

        const stat = await fs.stat(finalPath);
        console.log(
            '[Server Backup] Saved: '
                + filename
                + ' ('
                + (stat.size / 1024 / 1024).toFixed(1)
                + ' MB)',
        );
        res.write(JSON.stringify({
            type: 'done',
            ok: true,
            filename,
            size: stat.size,
            dir: backupsDir,
        }) + '\n');
        res.end();
    } catch (error) {
        // Wait for the file descriptor to close before the finally block
        // unlinks a partial archive; Windows cannot unlink an open file.
        await destroyBackupWritable(writeStream);
        if (abortTracker.signal.aborted || error?.name === 'AbortError') return;
        if (!res.headersSent) {
            if (error instanceof BackupSourceError) {
                res.status(error.statusCode || 500).json(backupSourceErrorPayload(error));
            } else {
                next(error);
            }
        } else if (!res.writableEnded) {
            res.write(JSON.stringify({
                type: 'error',
                message: error?.message || 'server backup failed',
                ...(error instanceof BackupSourceError ? { code: error.code } : {}),
            }) + '\n');
            res.end();
        }
    } finally {
        if (tmpPath) await fs.unlink(tmpPath).catch(() => {});
        await backupSource?.close();
        abortTracker.cleanup();
    }
});
